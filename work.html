import math
import random
from typing import List, Tuple

class FractalGenerator:
    """
    分形图案生成器
    使用递归算法生成自然形态的分形图案
    支持多种分形类型：树、蕨类、雪花等
    """
    
    def __init__(self, depth=5, canvas_width=800, canvas_height=600):
        self.depth = depth
        self.canvas_width = canvas_width
        self.canvas_height = canvas_height
        self.color_palette = ['#00BFFF', '#50fa7b', '#ff79c6', '#bd93f9']
        self.lines: List[Tuple] = []
    
    def generate_tree(self, x, y, size, angle, level):
        if level <= 0:
            return
        
        end_x = x + size * math.cos(angle)
        end_y = y + size * math.sin(angle)
        
        # 记录线条用于后续渲染
        self.lines.append((x, y, end_x, end_y, level))
        
        # 递归生成左右子分支
        branch_factor = 0.7 + random.uniform(-0.1, 0.1)
        angle_offset = 0.3 + random.uniform(-0.1, 0.1)
        
        self.generate_tree(end_x, end_y, size * branch_factor, angle + angle_offset, level - 1)
        self.generate_tree(end_x, end_y, size * branch_factor, angle - angle_offset, level - 1)
        
        # 添加随机第三个分支
        if random.random() > 0.7 and level > 2:
            self.generate_tree(end_x, end_y, size * branch_factor * 0.8, angle + 0.5, level - 2)
    
    def generate_fractal(self, fractal_type='tree'):
        """生成指定类型的分形图案"""
        self.lines.clear()
        
        if fractal_type == 'tree':
            start_x = self.canvas_width // 2
            start_y = self.canvas_height - 50
            self.generate_tree(start_x, start_y, 100, -math.pi/2, self.depth)
        
        return self.lines
